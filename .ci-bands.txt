
export interface SpectrogramBand {
  /** Quantised opacity shared by every cell in this band. */
  opacity: number;
  /** One SVG path covering every cell at that opacity. */
  d: string;
}

/**
 * The same spectrogram, but collapsed into one path per opacity band instead of
 * one rect per cell.
 *
 * This matters more than it looks. A 48x14 grid is up to 672 elements, and a
 * page showing twelve of them was serialising roughly half a megabyte of markup
 * plus an even larger React payload describing the identical elements a second
 * time. Quantising opacity into a few bands and emitting one path each drops
 * that to single figures of elements with no visible difference, because the
 * eye cannot resolve 0.61 from 0.63 opacity on an 84px tall figure anyway.
 */
export function spectrogramBands(
  sig: SoundSignature,
  seed: string,
  cols = 48,
  rows = 14,
  bands = 6,
): SpectrogramBand[] {
  const cells = spectrogram(sig, seed, cols, rows);
  const buckets = new Map<number, string[]>();

  for (const cell of cells) {
    // Snap to the nearest band, never to zero, so faint detail survives.
    const step = Math.max(1, Math.round(cell.v * bands));
    const parts = buckets.get(step) ?? [];
    parts.push(`M${cell.x} ${cell.y}h${cell.w}v${cell.h}h-${cell.w}Z`);
    buckets.set(step, parts);
  }

  return [...buckets.entries()]
    .sort((a, b) => a[0] - b[0])
    .map(([step, parts]) => ({
      opacity: Number((step / bands).toFixed(2)),
      d: parts.join(""),
    }));
}
